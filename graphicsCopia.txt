// sprites 4x5, por lo menos los numeros

int drawNumbers(SDL_Renderer *renderer, int *codigos) {
  // pixeles 20x20
  // la funcion recibira los codigos del numero correspondiente
  int indexC = 0;

  char *bin = hexa_a_bin(codigos[indexC]);
  char *bin2 = hexa_a_bin(codigos[indexC]);
  int num = 0;

  int positionX = 40; // posicion inicial
  int positionY = 40;

  bool salto = false;

  for (int x = 0; x < 6; x++) {

    for (int y = 0; y < strlen(bin2); y++) {

      num = bin[y] - '0'; // convertir a numero

      if (num > 0) {
        // DIBUJAR
        SDL_FRect square = {positionX, positionY, 20, 20};

        SDL_SetRenderDrawColor(renderer, 0, 255, 0, SDL_ALPHA_OPAQUE);

        SDL_RenderFillRect(renderer, &square);

        positionX += 20;
      } else {
        // SALTAR

        positionX += 20;
      }

      if (y == 4) {
        indexC++;
        salto = true;
        bin = hexa_a_bin(codigos[indexC]);

        // cambiar el binario
      }

      if (salto == true) {
        positionX = 40;
        positionY += 20;
        salto = false;
      }
    }
  }

  return 0;
}

int graphics(void) {

  int codigos[5] = {0xF0, 0X90, 0X90, 0X90, 0XF0};

  // Manejar error por si sdl falla en iniciar
  if (!SDL_Init(SDL_INIT_VIDEO)) {
    printf("Error: %s\n", SDL_GetError());
    return 1;
  }

  SDL_Window *window = SDL_CreateWindow("CHIP8", 600, 600, 0);

  if (!window) {
    printf("Error creando ventana: %s\n", SDL_GetError());
    SDL_Quit();
    return 1;
  }

  bool running = true;

  SDL_Event event;

  SDL_Renderer *renderer = SDL_CreateRenderer(window, NULL);

  while (running) {

    while (SDL_PollEvent(&event)) {

      SDL_SetRenderDrawColor(renderer, 0, 0, 0, 0);

      SDL_RenderClear(renderer);

      drawNumbers(renderer, codigos);

      SDL_RenderPresent(renderer);

      if (event.type == SDL_EVENT_QUIT) {
        running = false;
      }
    }
  }

  SDL_DestroyWindow(window);
  SDL_DestroyRenderer(renderer);

  SDL_Quit();

  return 0;
